cmake_minimum_required(VERSION 3.25)
project(nox_mac_startup_tests LANGUAGES CXX)
enable_testing()
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(core "${root}/noxcore")
add_executable(startup_tests startup_tests.cpp "${root}/macos/startup.cpp" "${root}/macos/user_data.cpp"
    "${core}/src/io/game_dir.cpp" "${core}/src/io/file.cpp")
target_compile_features(startup_tests PRIVATE cxx_std_20)
target_include_directories(startup_tests PRIVATE "${root}/macos" "${core}/include")
add_test(NAME mac_startup COMMAND startup_tests)
add_executable(user_data_tests user_data_tests.cpp "${root}/macos/user_data.cpp" "${core}/src/io/file.cpp")
target_compile_features(user_data_tests PRIVATE cxx_std_20)
target_include_directories(user_data_tests PRIVATE "${root}/macos" "${core}/include")
add_test(NAME mac_user_data COMMAND user_data_tests)
